import discord
from discord.ext import commands
import asyncio
import http.server
import socketserver
import threading
import os

# --- חלק 1: השרת שמרצה את RENDER (כדי שיהיה בחינם) ---
def run_on_render():
    PORT = int(os.environ.get("PORT", 8080))
    handler = http.server.SimpleHTTPRequestHandler
    with socketserver.TCPServer(("", PORT), handler) as httpd:
        print(f"✅ Render server serving at port {PORT}")
        httpd.serve_forever()

threading.Thread(target=run_on_render, daemon=True).start()

# --- חלק 2: הבוט של דיסקורד עם פקודות מהסרטון ---
intents = discord.Intents.default()
intents.message_content = True
bot = commands.Bot(command_prefix='!', intents=intents)

user_credits = {}
admin_ids = [1234567890] # תחליף את המספר הזה ב-ID שלך בדיסקורד!

@bot.event
async def on_ready():
    print(f'🚀 Bot {bot.user.name} is ONLINE!')

@bot.command()
async def setup(ctx):
    embed = discord.Embed(title="🚀 SMS Bomber Cloud V6", color=0xff0000)
    embed.add_field(name="פקודות", value="`!bomb [phone]` - התחלת הפצצה\n`!credits` - בדיקת יתרה", inline=False)
    await ctx.send(embed=embed)

@bot.command()
async def credits(ctx):
    balance = user_credits.get(ctx.author.id, 0)
    await ctx.send(f"💰 {ctx.author.name}, יש לך {balance} קרדיטים.")

@bot.command()
async def add_credits(ctx, member: discord.Member, amount: int):
    if ctx.author.id in admin_ids:
        user_credits[member.id] = user_credits.get(member.id, 0) + amount
        await ctx.send(f"✅ נוספו {amount} קרדיטים ל-{member.display_name}.")
    else:
        await ctx.send("❌ אתה לא אדמין!")

@bot.command()
async def bomb(ctx, phone: str):
    balance = user_credits.get(ctx.author.id, 0)
    if balance <= 0:
        await ctx.send("❌ אין לך קרדיטים!")
        return
    
    await ctx.send(f"💣 מתחיל מתקפה על {phone}...")
    user_credits[ctx.author.id] -= 1
    await asyncio.sleep(5)
    await ctx.send(f"✅ הסתיים! נשאר לך {user_credits[ctx.author.id]} קרדיטים.")

# משיכת הטוקן מהגדרות Render
token = os.environ.get("DISCORD_TOKEN")
if token:
    bot.run(token)
else:
    print("❌ ERROR: No DISCORD_TOKEN found in Environment Variables!")
