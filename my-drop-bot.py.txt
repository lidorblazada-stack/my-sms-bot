# --- שרת פנימי למניעת קריסה ב-Render ---
def run_on_render():
    PORT = int(os.environ.get("PORT", 8080))
    handler = http.server.SimpleHTTPRequestHandler
    with socketserver.TCPServer(("", PORT), handler) as httpd:
        httpd.serve_forever()

threading.Thread(target=run_on_render, daemon=True).start()

class MyBot(commands.Bot):

        intents = discord.Intents.default()
        intents.members = True
        super().__init__(command_prefix="!", intents=intents)

    async def setup_hook(self):
        await self.tree.sync()

bot = MyBot()

@bot.tree.command(name="bomb", description="הפעלת הפצצה")
async def bomb(interaction: discord.Interaction, phone: str):
    is_lifetime = interaction.user.id in lifetime_users
    balance = user_credits.get(interaction.user.id, 0)
    
    if not is_lifetime and balance <= 0:
        await interaction.response.send_message("❌ אין לך קרדיטים!", ephemeral=True)
        return
    
    await interaction.response.send_message(f"💣 מתחיל הפצצה על {phone}...")
    if not is_lifetime:
        user_credits[interaction.user.id] -= 1
    await asyncio.sleep(5)
    await interaction.followup.send(f"✅ ההפצצה על {phone} הסתיימה!")

@bot.tree.command(name="add_credits", description="הוספת קרדיטים (אדמין בלבד)")
async def add_credits(interaction: discord.Interaction, member: discord.Member, amount: int):
    if not any(role.name == ADMIN_ROLE_NAME for role in interaction.user.roles):
        await interaction.response.send_message("❌ אין לך הרשאה!", ephemeral=True)
        return
    user_credits[member.id] = user_credits.get(member.id, 0) + amount
    await interaction.response.send_message(f"✅ נוספו {amount} קרדיטים ל-{member.mention}.")

@bot.tree.command(name="set_lifetime", description="הגדרת לייפטיים (אדמין בלבד)")
async def set_lifetime(interaction: discord.Interaction, member: discord.Member, status: bool):
    if not any(role.name == ADMIN_ROLE_NAME for role in interaction.user.roles):
        await interaction.response.send_message("❌ אין לך הרשאה!", ephemeral=True)
        return
    if status: lifetime_users.add(member.id)
    else: lifetime_users.discard(member.id)
    await interaction.response.send_message(f"♾️ סטטוס לייפטיים עודכן ל-{member.mention}.")
