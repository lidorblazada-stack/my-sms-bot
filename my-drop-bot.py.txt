import discord

import json

# --- ניהול מסד נתונים פשוט ---
DB_FILE = "credits_db.json"

def load_data():
    if os.path.exists(DB_FILE):
        with open(DB_FILE, "r") as f:
            data = json.load(f)
            credits = {int(k): v for k, v in data.get("credits", {}).items()}
            lifetime = set(data.get("lifetime", []))
            return credits, lifetime
    return {}, set()

def save_data(credits, lifetime):
    with open(DB_FILE, "w") as f:
        json.dump({"credits": credits, "lifetime": list(lifetime)}, f)

user_credits, lifetime_users = load_data()

# --- שרת פנימי ל-Render ---
def run_on_render():
    PORT = int(os.environ.get("PORT", 8080))
    handler = http.server.SimpleHTTPRequestHandler
    with socketserver.TCPServer(("", PORT), handler) as httpd:
        httpd.serve_forever()
threading.Thread(target=run_on_render, daemon=True).start()

class MyBot(commands.Bot):
    def __init__(self):
        intents = discord.Intents.default()
        intents.members = True
        super().__init__(command_prefix="!", intents=intents)
    async def setup_hook(self):
        await self.tree.sync()

bot = MyBot()
ADMIN_ROLE_NAME = "Owner" # שם הרול המורשה

# --- Modal (החלון הקופץ) ---
class SpamModal(discord.ui.Modal, title='Spam-Me Spam Control'):
    phone = discord.ui.TextInput(label='Phone Number', placeholder='05XXXXXXXX', required=True)
    credits_to_use = discord.ui.TextInput(label='Credits to use', placeholder='1 Credit = 35 seconds', required=True)

    async def on_submit(self, interaction: discord.Interaction):
        try:
            amount = int(self.credits_to_use.value)
            total_seconds = amount * 35
            
            is_lifetime = interaction.user.id in lifetime_users
            balance = user_credits.get(interaction.user.id, 0)
            
            if not is_lifetime and balance < amount:
                await interaction.response.send_message(f"❌ אין לך מספיק קרדיטים! (יתרה: {balance})", ephemeral=True)
                return

            if not is_lifetime:
                user_credits[interaction.user.id] -= amount
                save_data(user_credits, lifetime_users)
            
            await interaction.response.send_message(f"💣 הפצצה על **{self.phone.value}** החלה! משך: **{total_seconds}** שניות.")
            await asyncio.sleep(5) 
            await interaction.followup.send(f"✅ הסתיימה ההפצצה על {self.phone.value}!")
            
        except ValueError:
            await interaction.response.send_message("❌ נא להזין מספר תקין.", ephemeral=True)

        await interaction.response.send_message(f"💰 יתרה נוכחית: **{balance}**", ephemeral=True)

@bot.tree.command(name="add_credits", description="הוספת קרדיטים (ל-Owner בלבד)")
async def add_credits(interaction: discord.Interaction, member: discord.Member, amount: int):
    if not any(role.name == ADMIN_ROLE_NAME for role in interaction.user.roles):
        await interaction.response.send_message("❌ אין לך הרשאה!", ephemeral=True)
        return
    user_credits[member.id] = user_credits.get(member.id, 0) + amount
    save_data(user_credits, lifetime_users)
    await interaction.response.send_message(f"✅ נוספו {amount} קרדיטים ל-{member.mention}.")

@bot.tree.command(name="remove_credits", description="הסרת קרדיטים (ל-Owner בלבד)")
async def remove_credits(interaction: discord.Interaction, member: discord.Member, amount: int):
    if not any(role.name == ADMIN_ROLE_NAME for role in interaction.user.roles):
        await interaction.response.send_message("❌ אין לך הרשאה!", ephemeral=True)
        return
    current = user_credits.get(member.id, 0)
    user_credits[member.id] = max(0, current - amount)
    save_data(user_credits, lifetime_users)
    await interaction.response.send_message(f"📉 הוסרו {amount} קרדיטים מ-{member.mention}.")

@bot.tree.command(name="set_lifetime", description="הגדרת לייפטיים (ל-Owner בלבד)")
async def set_lifetime(interaction: discord.Interaction, member: discord.Member, status: bool):
    if not any(role.name == ADMIN_ROLE_NAME for role in interaction.user.roles):
        await interaction.response.send_message("❌ אין לך הרשאה!", ephemeral=True)
        return
    if status: lifetime_users.add(member.id)
    else: lifetime_users.discard(member.id)
    save_data(user_credits, lifetime_users)
    await interaction.response.send_message(f"♾️ סטטוס לייפטיים עבור {member.mention} עודכן.")
