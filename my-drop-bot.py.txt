import discord
from discord import app_commands
from discord.ext import commands
import asyncio
import http.server
import socketserver
import threading
import os

# --- חלק 1: שרת למניעת קריסה ב-Render (חינם) ---
def run_on_render():
    PORT = int(os.environ.get("PORT", 8080))
    handler = http.server.SimpleHTTPRequestHandler
    with socketserver.TCPServer(("", PORT), handler) as httpd:
        print(f"✅ Render server active on port {PORT}")
        httpd.serve_forever()

threading.Thread(target=run_on_render, daemon=True).start()

# --- חלק 2: הגדרות הבוט ופקודות סלאש ---
class MyBot(commands.Bot):
    def __init__(self):
        intents = discord.Intents.default()
        intents.members = True 
        intents.message_content = True
        super().__init__(command_prefix="!", intents=intents)

    async def setup_hook(self):
        await self.tree.sync()
        print(f"✅ Slash commands synced!")

bot = MyBot()
user_credits = {}
ADMIN_ROLE_NAME = "Owner" 

@bot.tree.command(name="setup", description="הצגת תפריט הבוט")
async def setup(interaction: discord.Interaction):
    embed = discord.Embed(title="🚀 SMS Bomber Cloud V6", color=0xff0000)
    embed.add_field(name="שימוש", value="`/bomb` - שליחת הודעות\n`/credits` - בדיקת יתרה", inline=False)
    await interaction.response.send_message(embed=embed)

@bot.tree.command(name="add_credits", description="הוספת קרדיטים (לבעלי רול Owner בלבד)")
async def add_credits(interaction: discord.Interaction, member: discord.Member, amount: int):
    if not any(role.name == ADMIN_ROLE_NAME for role in interaction.user.roles):
        await interaction.response.send_message(f"❌ רק לבעלי רול **{ADMIN_ROLE_NAME}** מותר!", ephemeral=True)
        return
    user_credits[member.id] = user_credits.get(member.id, 0) + amount
    await interaction.response.send_message(f"✅ נוספו {amount} קרדיטים ל-{member.display_name}.")

@bot.tree.command(name="bomb", description="הפעלת הפצצת SMS")
async def bomb(interaction: discord.Interaction, phone: str):
    balance = user_credits.get(interaction.user.id, 0)
    if balance <= 0:
        await interaction.response.send_message("❌ אין לך קרדיטים!", ephemeral=True)
        return
    await interaction.response.send_message(f"💣 מתחיל מתקפה על {phone}...")
    user_credits[interaction.user.id] -= 1
    await asyncio.sleep(5)
    await interaction.followup.send(f"✅ המתקפה על {phone} הסתיימה!")

# הרצה (הטוקן נמשך אוטומטית מ-Render)
token = os.environ.get("DISCORD_TOKEN")
bot.run(token)
