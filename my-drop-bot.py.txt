import discord
from discord import app_commands
from discord.ext import commands
import asyncio
import os
import http.server
import socketserver
import threading

# --- שרת פנימי למניעת קריסה ב-Render ---
def run_on_render():
    PORT = int(os.environ.get("PORT", 8080))
    handler = http.server.SimpleHTTPRequestHandler
    with socketserver.TCPServer(("", PORT), handler) as httpd:
        httpd.serve_forever()

threading.Thread(target=run_on_render, daemon=True).start()

class MyBot(commands.Bot):
    def __init__(self):
        intents = discord.Intents.default()
        intents.members = True
        intents.message_content = True
        super().__init__(command_prefix="!", intents=intents)

    async def setup_hook(self):
        await self.tree.sync()
        print(f"✅ כל הפקודות סונכרנו בהצלחה!")

bot = MyBot()
user_credits = {}
lifetime_users = set() 
ADMIN_ROLE_NAME = "Owner" # הרול מהתמונה שלך

# --- פקודות משתמש רגיל ---

@bot.tree.command(name="setup", description="הצגת תפריט הבוט המלא (V6 Platinum)")
async def setup(interaction: discord.Interaction):
    embed = discord.Embed(title="🚀 VOUGE SPAM-ME V6 🚀", color=0x5865F2)
    embed.add_field(name="💣 התקפה", value="`/bomb` - שליחת הפצצת SMS\n`/layfight` - סטטוס מערכת", inline=False)
    embed.add_field(name="💰 חשבון", value="`/credits` - יתרת קרדיטים\n`/info` - מידע על המערכת", inline=False)
    embed.set_footer(text="Developed by Lidor • V6.2")
    await interaction.response.send_message(embed=embed)

@bot.tree.command(name="credits", description="בדיקת יתרת הקרדיטים שלך")
async def credits(interaction: discord.Interaction):
    if interaction.user.id in lifetime_users:
        await interaction.response.send_message("💰 יתרה: **Unlimited (Lifetime) ♾️**", ephemeral=True)
    else:
        balance = user_credits.get(interaction.user.id, 0)
        await interaction.response.send_message(f"💰 יתרה נוכחית: **{balance}** קרדיטים.", ephemeral=True)

@bot.tree.command(name="bomb", description="הפעלת הפצצת SMS על מספר")
async def bomb(interaction: discord.Interaction, phone: str):
    is_lifetime = interaction.user.id in lifetime_users
    balance = user_credits.get(interaction.user.id, 0)
    
    if not is_lifetime and balance <= 0:
        await interaction.response.send_message("❌ אין לך קרדיטים! פנה למנהל לרכישה.", ephemeral=True)
        return
    
    await interaction.response.send_message(f"💣 מתחיל התקפה על **{phone}**...")
    if not is_lifetime:
        user_credits[interaction.user.id] -= 1
    
    await asyncio.sleep(5)
    await interaction.followup.send(f"✅ ההתקפה על {phone} הושלמה בהצלחה!")

# --- פקודות ADMIN (מוגנות ברול Owner) ---

@bot.tree.command(name="add_credits", description="[ADMIN] הוספת קרדיטים למשתמש")
async def add_credits(interaction: discord.Interaction, member: discord.Member, amount: int):
    if not any(role.name == ADMIN_ROLE_NAME for role in interaction.user.roles):
        await interaction.response.send_message(f"❌ רק לבעלי רול **{ADMIN_ROLE_NAME}** מותר לעשות זאת!", ephemeral=True)
        return
    user_credits[member.id] = user_credits.get(member.id, 0) + amount
    await interaction.response.send_message(f"✅ נוספו {amount} קרדיטים ל-{member.mention}.")

@bot.tree.command(name="set_lifetime", description="[ADMIN] הגדרת משתמש כ-Lifetime")
async def set_lifetime(interaction: discord.Interaction, member: discord.Member, status: bool):
    if not any(role.name == ADMIN_ROLE_NAME for role in interaction.user.roles):
        await interaction.response.send_message(f"❌ רק לבעלי רול **{ADMIN_ROLE_NAME}** מותר לעשות זאת!", ephemeral=True)
        return
    if status:
        lifetime_users.add(member.id)
        await interaction.response.send_message(f"♾️ {member.mention} הוגדר כ-**Lifetime**!")
    else:
        lifetime_users.discard(member.id)
        await interaction.response.send_message(f"❌ סטטוס Lifetime הוסר מ-{member.mention}.")
